import os
import telebot
from moviepy.editor import VideoFileClip
from flask import Flask
from threading import Thread

# Web server Render o'chirib qo'ymasligi uchun
app = Flask('')

@app.route('/')
def home():
    return "Bot faol ishlamoqda!"

def run():
    app.run(host='0.0.0.0', port=8080)

def keep_alive():
    t = Thread(target=run)
    t.start()

# Telegram Bot Tokeningizni shu yerga yozing:
TOKEN = 'YOUR_BOT_TOKEN_HERE'
bot = telebot.TeleBot(TOKEN)

@bot.message_handler(commands=['start'])
def start_message(message):
    bot.reply_to(message, "Salom! Menga video yuboring, men undan audiosini ajratib beraman.")

@bot.message_handler(content_types=['video'])
def handle_video(message):
    try:
        msg = bot.reply_to(message, "Video yuklanmoqda va qayta ishlanmoqda, kuting...")
        file_info = bot.get_file(message.video.file_id)
        downloaded_file = bot.download_file(file_info.file_path)
        
        video_path = "temp_video.mp4"
        audio_path = "result_audio.mp3"

        with open(video_path, 'wb') as new_file:
            new_file.write(downloaded_file)

        # Videodan audioni ajratib olish
        clip = VideoFileClip(video_path)
        clip.audio.write_audiofile(audio_path)
        clip.close()

        # Audioni foydalanuvchiga yuborish
        with open(audio_path, 'rb') as audio:
            bot.send_audio(message.chat.id, audio)

        # Vaqtinchalik fayllarni o'chirish
        if os.path.exists(video_path):
            os.remove(video_path)
        if os.path.exists(audio_path):
            os.remove(audio_path)

    except Exception as e:
        bot.reply_to(message, f"Xatolik yuz berdi: {str(e)}")

if __name__ == '__main__':
    keep_alive()
    bot.polling(none_stop=True)
